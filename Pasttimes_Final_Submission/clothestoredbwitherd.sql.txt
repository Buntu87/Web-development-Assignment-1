-- ==============================
-- Database
-- ==============================
CREATE DATABASE IF NOT EXISTS clothestoredbwitherd;
USE clothestoredbwitherd;

-- ==============================
-- USERS (Admins + Customers)
-- ==============================
CREATE TABLE Users (
    UserID INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    Email VARCHAR(100) UNIQUE NOT NULL,
    Password VARCHAR(255) NOT NULL,
    Role ENUM('admin','customer') DEFAULT 'customer'
);

-- ==============================
-- SIDEBAR MODULES (for dashboard menu)
-- ==============================
CREATE TABLE SidebarMenu (
    MenuID INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    Link VARCHAR(255) NOT NULL,
    Icon VARCHAR(100)
);

-- ==============================
-- SELLERS (for products section)
-- ==============================
CREATE TABLE Sellers (
    SellerID INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(100) NOT NULL,
    ContactInfo VARCHAR(255)
);

-- ==============================
-- PRODUCTS (cards in dashboard)
-- ==============================
CREATE TABLE Products (
    ProductID INT AUTO_INCREMENT PRIMARY KEY,
    Name VARCHAR(150) NOT NULL,
    Price DECIMAL(10,2) NOT NULL,
    Stock INT NOT NULL,
    Image VARCHAR(255),
    SellerID INT,
    FOREIGN KEY (SellerID) REFERENCES Sellers(SellerID)
    ON DELETE SET NULL
    ON UPDATE CASCADE
);

-- ==============================
-- ORDERS (dashboard table + stats card)
-- ==============================
CREATE TABLE Orders (
    OrderID INT AUTO_INCREMENT PRIMARY KEY,
    UserID INT,
    OrderDate DATETIME DEFAULT CURRENT_TIMESTAMP,
    Status ENUM('pending','processing','shipped','completed','cancelled') DEFAULT 'pending',
    TotalAmount DECIMAL(10,2),
    FOREIGN KEY (UserID) REFERENCES Users(UserID)
    ON DELETE CASCADE
);

-- ==============================
-- ORDER ITEMS (cart system)
-- ==============================
CREATE TABLE OrderItems (
    OrderItemID INT AUTO_INCREMENT PRIMARY KEY,
    OrderID INT,
    ProductID INT,
    Quantity INT NOT NULL,
    Price DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (OrderID) REFERENCES Orders(OrderID)
    ON DELETE CASCADE,
    FOREIGN KEY (ProductID) REFERENCES Products(ProductID)
    ON DELETE CASCADE
);

-- ==============================
-- MESSAGES (admin panel inbox)
-- ==============================
CREATE TABLE Messages (
    MessageID INT AUTO_INCREMENT PRIMARY KEY,
    SenderID INT,
    ReceiverID INT,
    Content TEXT,
    Timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (SenderID) REFERENCES Users(UserID)
    ON DELETE CASCADE,
    FOREIGN KEY (ReceiverID) REFERENCES Users(UserID)
    ON DELETE CASCADE
);

-- ==============================
-- DASHBOARD STATS VIEW (for cards in UI)
-- ==============================
CREATE VIEW DashboardStats AS
SELECT 
    (SELECT COUNT(*) FROM Users) AS TotalUsers,
    (SELECT COUNT(*) FROM Products) AS TotalProducts,
    (SELECT COUNT(*) FROM Orders) AS TotalOrders,
    (SELECT IFNULL(SUM(TotalAmount),0) FROM Orders) AS TotalRevenue;
